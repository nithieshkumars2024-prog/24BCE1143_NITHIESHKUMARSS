   ID int NOT NULL,
   LastName varchar(255) UNIQUE NOT NULL,
   FirstName varchar(255) NOT NULL,
   Age int
);

-- Dropping for next constraints adding
DROP TABLE Persons;

-- Adding check constraints
CREATE TABLE Persons(
   ID int NOT NULL,
   LastName varchar(255) UNIQUE NOT NULL,
   FirstName varchar(255) NOT NULL,
   Age int NOT NULL,
   CHECK (Age>=18 AND Age<=23)
);

-- Adding check constraint after creating(My own constraint)
ALTER TABLE Persons
ADD CHECK (ID>=0 AND ID<=10000);

DESC Persons;

-- Understanding default constraint(values which are default if we do not mention that column's value)
CREATE TABLE Persons1(
   ID int NOT NULL,
   LastName varchar(255) UNIQUE NOT NULL,
   FirstName varchar(255) NOT NULL,
   Age int NOT NULL,
   City VARCHAR(255) DEFAULT 'Coimbatore',
   CHECK (Age>=18 AND Age<=23)
);

DESC Persons1;

CREATE TABLE Category(
   cid INT PRIMARY KEY,
   cname VARCHAR(100) NOT NULL
);

DESC Category;

INSERT INTO Category VALUES (101,'electronics'),(102,'Home App');

CREATE TABLE Products(
   pid INT PRIMARY KEY,
   pname VARCHAR(100) NOT NULL,
   cid INT,
   CONSTRAINT fk_cid  -- Constraint constraint_name(helps to drop later)
   FOREIGN KEY(cid) REFERENCES category(cid)
);

DESC Products;

-- The foreign key can only take values which are present in parent table(duplicated or null also allowed but not the keys not present in the table.)
