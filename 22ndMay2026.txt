create table departments(
did int primary key,
dname varchar(25) not null unique
);

select * from departments order by did;

create table HOD(
Hid int primary key,
hname varchar(25) not null unique
);

insert into HOD values 
(101,'Arun'),(102,'Jayanth'),(104,'Karthik'),(105,'Murali');

SELECT * FROM HOD;

SELECT * FROM 
departments CROSS JOIN HOD ORDER BY DID ASC;


SELECT * FROM 
departments INNER JOIN HOD WHERE departments.did = hod.hid;

SELECT * FROM 
departments NATURAL JOIN HOD;

SELECT * FROM 
departments LEFT OUTER JOIN HOD ON (departments.did = hod.hid) ORDER BY DID;

SELECT * FROM 
departments FULL JOIN HOD ON (departments.did = hod.hid);
